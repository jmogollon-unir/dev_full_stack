#!/bin/bash
# =============================================================================
# Relatos de Papel - User data para la EC2 (Amazon Linux 2023, x86_64)
# Se pega en: EC2 > Launch instance > Advanced details > User data
# Se ejecuta UNA vez, como root, en el primer arranque de la instancia.
# =============================================================================

# ---------------------- EDITA SOLO ESTE BLOQUE -------------------------------
DB_HOST="REEMPLAZAR-por-el-endpoint-de-RDS.xxxxxxxx.us-east-1.rds.amazonaws.com"
DB_PORT="3306"
DB_NAME="relatos_papel"
DB_USER="root"
DB_PASSWORD="mysql"

# Opcional: repositorio Git (HTTPS, público) con este proyecto. Si lo dejas vacío, el script
# instala todo lo necesario y espera a que subas el proyecto por scp (ver docs/AWS-paso-a-paso.md).
REPO_URL=""
REPO_BRANCH="main"
# -----------------------------------------------------------------------------

set -uo pipefail
exec > >(tee -a /var/log/user-data-init.log) 2>&1
echo "===== [$(date)] Inicio user-data Relatos de Papel ====="

export HOME=/root

# 1) Paquetes: Java 21 (JDK para compilar), git, cliente MySQL (diagnóstico) y herramientas
dnf install -y java-21-amazon-corretto-devel git unzip tar mariadb105

# 2) Swap de 2 GB (la compilación del front con Node necesita memoria en instancias pequeñas)
if [ ! -f /swapfile ]; then
  dd if=/dev/zero of=/swapfile bs=1M count=2048
  chmod 600 /swapfile
  mkswap /swapfile
  swapon /swapfile
  echo "/swapfile none swap sw 0 0" >> /etc/fstab
fi

# 3) Variables de entorno de la aplicación (para cambiar DB_HOST después: editar este archivo
#    y ejecutar "sudo systemctl restart relatos-papel")
cat > /etc/relatos-papel.env <<ENVEOF
SERVER_PORT=80
DB_HOST=${DB_HOST}
DB_PORT=${DB_PORT}
DB_NAME=${DB_NAME}
DB_USER=${DB_USER}
DB_PASSWORD=${DB_PASSWORD}
DB_USE_SSL=false
ENVEOF
chmod 600 /etc/relatos-papel.env

# 4) Servicio systemd (arranca solo al reiniciar la instancia; escucha en el puerto 80)
mkdir -p /opt/relatos-papel
chown ec2-user:ec2-user /opt/relatos-papel
cat > /etc/systemd/system/relatos-papel.service <<'SVCEOF'
[Unit]
Description=Relatos de Papel (Spring Boot: API + front)
After=network-online.target
Wants=network-online.target

[Service]
User=ec2-user
WorkingDirectory=/opt/relatos-papel
EnvironmentFile=/etc/relatos-papel.env
ExecStart=/usr/bin/java -Xms256m -Xmx768m -jar /opt/relatos-papel/app.jar
Restart=on-failure
RestartSec=15
# Permite a un usuario sin privilegios escuchar en el puerto 80
AmbientCapabilities=CAP_NET_BIND_SERVICE
CapabilityBoundingSet=CAP_NET_BIND_SERVICE

[Install]
WantedBy=multi-user.target
SVCEOF
systemctl daemon-reload
systemctl enable relatos-papel

# 5) Script de despliegue: compila (front + back en un solo JAR) y reinicia el servicio
#    Uso:  sudo /opt/relatos-papel/deploy.sh                       -> git pull + compilar (si hay REPO_URL)
#          sudo /opt/relatos-papel/deploy.sh /home/ec2-user/relatos-papel.zip   -> descomprime el zip + compilar
cat > /opt/relatos-papel/deploy.sh <<'DEPEOF'
#!/bin/bash
set -euo pipefail
export HOME=/root
export JAVA_HOME="$(dirname "$(dirname "$(readlink -f "$(which java)")")")"
export MAVEN_OPTS="-Xmx1g"
SRC=/opt/relatos-papel/src
ZIP="${1:-}"

if [ -n "$ZIP" ]; then
  rm -rf "$SRC" && mkdir -p "$SRC"
  unzip -q "$ZIP" -d "$SRC"
  # si el zip trae una carpeta raíz (relatos-papel/), se sube un nivel
  if [ ! -f "$SRC/pom.xml" ]; then
    INNER="$(find "$SRC" -maxdepth 2 -name pom.xml -printf '%h\n' | head -1)"
    mv "$INNER"/* "$INNER"/.[!.]* "$SRC"/ 2>/dev/null || true
  fi
elif [ -d "$SRC/.git" ]; then
  git -C "$SRC" pull
else
  echo "No hay código en $SRC. Indica un zip: sudo $0 /home/ec2-user/relatos-papel.zip"; exit 1
fi

cd "$SRC"
chmod +x mvnw
./mvnw -B -DskipTests clean package        # compila el front (Node/Vite) y el back en UN solo JAR
JAR="$(find target -maxdepth 1 -name 'relatos-papel-*.jar' ! -name '*.original' | head -1)"
install -o ec2-user -g ec2-user -m 644 "$JAR" /opt/relatos-papel/app.jar
systemctl restart relatos-papel
echo "Despliegue terminado. Logs: journalctl -u relatos-papel -f"
DEPEOF
chmod +x /opt/relatos-papel/deploy.sh

# 6) Si se indicó un repositorio, se clona, compila y arranca. Si no, queda listo para scp.
if [ -n "${REPO_URL}" ]; then
  git clone --branch "${REPO_BRANCH}" "${REPO_URL}" /opt/relatos-papel/src
  /opt/relatos-papel/deploy.sh
else
  echo "REPO_URL vacío: sube el proyecto y ejecuta: sudo /opt/relatos-papel/deploy.sh /home/ec2-user/relatos-papel.zip"
fi

echo "===== [$(date)] Fin user-data ====="
